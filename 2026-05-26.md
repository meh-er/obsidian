# Monad
```
(>>=) :: Monad m => m a -> (a -> m b) -> m b (bind)
(>>) :: Monad m => m a -> m b -> m b (sequence, discard left)

return :: Monad m => a -> m a (wrap value in monad)

mapM :: Monad m => (a -> m b) -> [a] -> m [b] (map monadic function over a list) -- use in do blocks
	-- correct:
	children' <- mapM helper children
	return (Node l (x,s) children')
mapM_ :: Monad m => (a -> m b) -> [a] -> m () (like mapM but discards results)

sequence :: Monad m => [m a] -> m [a] (run a list of monadic actions)

when :: Applicative f => Bool -> f () -> f () (conditionally run an action)


STATE
get :: State s s (read current state)
put :: s -> State s () (replace state)
	-- use to reset stat when both branches need the same starting value (E.g. depth)
	s <- get
	modify (+1)
	l' <- helper l
	put s -- resets
	modify (+1)
	r' <- helper r
modify :: (s -> s) -> State s () (update state using function)
	modify (+1) -- correct
	modify (subtract 1) -- correct for decrement
	-- modify must have get before it
gets :: (s -> a) -> State s a

runState :: State s a -> s -> (a,s) (returns result and final state)
evalState :: State s a -> s -> a (return result only)
execState :: State s a -> s -> s (returns final state only)

WRITER
tell :: MonadWriter w m => w -> m () (append to log)
listen :: MonadWriter w m => m a -> m (a,w) (run action and capture its log)

runWriter :: Writer w a -> (a,w) (returns result and log)
execWriter :: Writer w a -> w (returns log only)
```
